cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(cloudtune_background LANGUAGES C CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
endif()
set(NCM_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
if(NOT TARGET netease_api)
  find_package(CURL REQUIRED)
  add_library(netease_api STATIC "${NCM_ROOT}/src/api.cpp" "${NCM_ROOT}/src/crypto.cpp")
  target_include_directories(netease_api PUBLIC "${NCM_ROOT}/include" "${NCM_ROOT}/vendor/nlohmann/include")
  if(NINTENDO_SWITCH)
    target_include_directories(netease_api PUBLIC ${CURL_INCLUDE_DIRS})
    target_link_directories(netease_api PUBLIC "$ENV{DEVKITPRO}/portlibs/switch/lib")
    target_link_libraries(netease_api PUBLIC CURL::libcurl mbedtls mbedx509 mbedcrypto z)
  else()
    find_path(MBEDTLS_INCLUDE_DIR mbedtls/aes.h REQUIRED)
    find_library(MBEDTLS_LIBRARY mbedtls REQUIRED)
    find_library(MBEDX509_LIBRARY mbedx509 REQUIRED)
    find_library(MBEDCRYPTO_LIBRARY mbedcrypto REQUIRED)
    target_include_directories(netease_api PUBLIC "${MBEDTLS_INCLUDE_DIR}")
    target_link_libraries(netease_api PUBLIC CURL::libcurl "${MBEDTLS_LIBRARY}" "${MBEDX509_LIBRARY}" "${MBEDCRYPTO_LIBRARY}" z)
  endif()
endif()
if(NOT TARGET netease_accounts)
  add_library(netease_accounts STATIC "${NCM_ROOT}/src/accounts.cpp")
  target_link_libraries(netease_accounts PUBLIC netease_api)
endif()
add_library(ncm_engine STATIC engine.cpp audio_decoder.cpp aac_decoder.cpp source/queue.c)
target_include_directories(ncm_engine PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${NCM_ROOT}/include")
target_link_libraries(ncm_engine PUBLIC netease_api netease_accounts)
if(NOT TARGET cloudtune_airplay)
  if(NINTENDO_SWITCH)
    set(MBEDTLS_INCLUDE_DIR "$ENV{DEVKITPRO}/portlibs/switch/include")
    set(MBEDCRYPTO_LIBRARY "$ENV{DEVKITPRO}/portlibs/switch/lib/libmbedcrypto.a")
  endif()
  add_subdirectory("${NCM_ROOT}/airplay" "${CMAKE_CURRENT_BINARY_DIR}/airplay")
endif()
target_link_libraries(ncm_engine PRIVATE cloudtune_audio_codecs)
add_library(ncm_runtime STATIC runtime.cpp)
target_link_libraries(ncm_runtime PUBLIC ncm_engine cloudtune_airplay)
if(NINTENDO_SWITCH)
  # Drop unused API/account helpers from the resident executable. Keep decoder
  # optimization unchanged; only section granularity and linker GC change.
  foreach(component netease_api netease_accounts ncm_engine ncm_runtime cloudtune_airplay)
    target_compile_options(${component} PRIVATE -ffunction-sections -fdata-sections)
  endforeach()
  target_compile_definitions(ncm_engine PUBLIC __SWITCH__)
  target_link_libraries(ncm_engine PUBLIC nx)
  add_library(ncm_ipc_client STATIC client/playback_client.c client/timed_ipc.c)
  target_include_directories(ncm_ipc_client PUBLIC "${NCM_ROOT}/include")
  target_link_libraries(ncm_ipc_client PUBLIC nx)
  set(NCM_STARTUP_SOURCES source/startup.c source/heap.c source/boot.c source/boot_query.c source/resources.c)
  add_executable(cloudtune-background source/main.cpp ${NCM_STARTUP_SOURCES} source/ipc_server.c)
  target_compile_definitions(cloudtune-background PRIVATE CLOUDTUNE_VERSION="${PROJECT_VERSION}")
  target_include_directories(cloudtune-background PRIVATE source)
  target_link_libraries(cloudtune-background PRIVATE ncm_runtime)
  target_compile_options(cloudtune-background PRIVATE -ffunction-sections -fdata-sections)
  target_link_options(cloudtune-background PRIVATE "-Wl,--gc-sections")
  # Native allocator probe is built/run explicitly; never part of an SD package.
  add_executable(ncm-switch-heap-probe EXCLUDE_FROM_ALL
    "${NCM_ROOT}/tests/switch_heap_probe.c" ${NCM_STARTUP_SOURCES})
  target_include_directories(ncm-switch-heap-probe PRIVATE source "${NCM_ROOT}/include")
  target_link_libraries(ncm-switch-heap-probe PRIVATE nx)
  target_link_options(ncm-switch-heap-probe PRIVATE "-Wl,--wrap=svcSetHeapSize" "-Wl,--wrap=ncmBootHasHomeMenu")
  add_executable(ncm-switch-runtime-probe EXCLUDE_FROM_ALL source/main.cpp ${NCM_STARTUP_SOURCES}
    source/ipc_server.c "${NCM_ROOT}/tests/switch_boot_boundary.c")
  target_include_directories(ncm-switch-runtime-probe PRIVATE source)
  target_compile_definitions(ncm-switch-runtime-probe PRIVATE CLOUDTUNE_VERSION="${PROJECT_VERSION}-probe")
  target_link_libraries(ncm-switch-runtime-probe PRIVATE ncm_runtime)
  target_link_options(ncm-switch-runtime-probe PRIVATE "-Wl,--wrap=ncmBootHasHomeMenu")
  find_program(NPDMTOOL npdmtool HINTS "$ENV{DEVKITPRO}/tools/bin" REQUIRED)
  find_program(ELF2NSO elf2nso HINTS "$ENV{DEVKITPRO}/tools/bin" REQUIRED)
  find_program(BUILD_PFS0 build_pfs0 HINTS "$ENV{DEVKITPRO}/tools/bin" REQUIRED)
  set(PACKAGE_DIR "${CMAKE_CURRENT_BINARY_DIR}/package")
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/cloudtune-background.nsp"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${PACKAGE_DIR}"
    COMMAND ${NPDMTOOL} "${CMAKE_CURRENT_SOURCE_DIR}/netease-sysmodule.json" "${PACKAGE_DIR}/main.npdm"
    COMMAND ${ELF2NSO} $<TARGET_FILE:cloudtune-background> "${PACKAGE_DIR}/main"
    COMMAND ${BUILD_PFS0} "${PACKAGE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/cloudtune-background.nsp"
    DEPENDS cloudtune-background "${CMAKE_CURRENT_SOURCE_DIR}/netease-sysmodule.json" VERBATIM)
  add_custom_target(cloudtune-background-package ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/cloudtune-background.nsp")
else()
  find_package(SDL2 REQUIRED)
  find_package(Threads REQUIRED)
  target_link_libraries(ncm_engine PUBLIC SDL2::SDL2 Threads::Threads)
  add_executable(ncm-playback-smoke tests/playback_smoke.cpp)
  target_link_libraries(ncm-playback-smoke PRIVATE ncm_engine)
  add_executable(ncm-decoder-tests tests/decoder_tests.cpp)
  target_link_libraries(ncm-decoder-tests PRIVATE ncm_engine)
  add_executable(ncm-flac-engine-tests tests/flac_engine_tests.cpp)
  target_link_libraries(ncm-flac-engine-tests PRIVATE ncm_engine)
  add_executable(ncm-live-playback tests/live_playback.cpp)
  target_link_libraries(ncm-live-playback PRIVATE ncm_engine)
  add_executable(ncm-queue-replace-test tests/queue_replace.cpp source/queue.c)
  target_include_directories(ncm-queue-replace-test PRIVATE "${NCM_ROOT}/include")
  target_link_options(ncm-queue-replace-test PRIVATE "-Wl,--wrap=rename")
endif()
