cmake_minimum_required(VERSION 3.16)
set(BOREALIS_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/vendor/borealis/library")
include(${BOREALIS_LIBRARY}/cmake/commonOption.cmake)
if(NOT PLATFORM_SWITCH AND NOT PLATFORM_DESKTOP)
  set(PLATFORM_DESKTOP ON CACHE BOOL "" FORCE)
endif()
include(${BOREALIS_LIBRARY}/cmake/toolchain.cmake)
project(CloudTuneNX VERSION 0.1.0 LANGUAGES C CXX)
option(CLOUDTUNE_BUILD_OVERLAY_PREVIEW "Build the Linux desktop overlay preview" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(PLATFORM_SWITCH)
  add_compile_definitions(__SWITCH__)
  add_compile_options(-march=armv8-a+crc+crypto -mtune=cortex-a57 -mtp=soft -ftls-model=local-exec)
  set(NINTENDO_SWITCH ON)
endif()
find_package(CURL REQUIRED)
find_package(Threads REQUIRED)
if(PLATFORM_SWITCH)
  # Never let a host .deps archive leak into a Switch link.
  set(MBEDTLS_INCLUDE_DIR "$ENV{DEVKITPRO}/portlibs/switch/include")
  set(MBEDTLS_LIBRARY "$ENV{DEVKITPRO}/portlibs/switch/lib/libmbedtls.a")
  set(MBEDX509_LIBRARY "$ENV{DEVKITPRO}/portlibs/switch/lib/libmbedx509.a")
  set(MBEDCRYPTO_LIBRARY "$ENV{DEVKITPRO}/portlibs/switch/lib/libmbedcrypto.a")
else()
  find_path(MBEDTLS_INCLUDE_DIR mbedtls/aes.h HINTS "${MBEDTLS_ROOT}/include" "${CMAKE_SOURCE_DIR}/.deps/mbedtls/include")
  find_library(MBEDTLS_LIBRARY mbedtls HINTS "${MBEDTLS_ROOT}/lib" "${CMAKE_SOURCE_DIR}/.deps/mbedtls/lib")
  find_library(MBEDX509_LIBRARY mbedx509 HINTS "${MBEDTLS_ROOT}/lib" "${CMAKE_SOURCE_DIR}/.deps/mbedtls/lib")
  find_library(MBEDCRYPTO_LIBRARY mbedcrypto HINTS "${MBEDTLS_ROOT}/lib" "${CMAKE_SOURCE_DIR}/.deps/mbedtls/lib")
endif()
if(NOT MBEDTLS_INCLUDE_DIR OR NOT MBEDCRYPTO_LIBRARY)
  message(FATAL_ERROR "mbedTLS 2.28 required. Run scripts/setup-host-deps.sh for an isolated host installation.")
endif()
add_library(netease_api STATIC src/api.cpp src/crypto.cpp)
target_include_directories(netease_api PUBLIC include vendor/nlohmann/include "${MBEDTLS_INCLUDE_DIR}")
target_link_libraries(netease_api PUBLIC CURL::libcurl ${MBEDTLS_LIBRARY} ${MBEDX509_LIBRARY} ${MBEDCRYPTO_LIBRARY} z Threads::Threads)
add_library(netease_library STATIC src/library.cpp)
target_link_libraries(netease_library PUBLIC netease_api)
add_subdirectory(airplay)
add_subdirectory(background)
if(CLOUDTUNE_BUILD_OVERLAY_PREVIEW)
  if(PLATFORM_SWITCH OR NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The overlay preview requires a native Linux build")
  endif()
  add_subdirectory(overlay/preview)
endif()
set(BRLS_RESOURCES_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
add_subdirectory(${BOREALIS_LIBRARY} ${CMAKE_BINARY_DIR}/borealis)
target_include_directories(borealis PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/nlohmann/include")
add_executable(CloudTune src/main.cpp src/model.cpp src/player.cpp src/ui.cpp vendor/qrcodegen/qrcodegen.cpp)
target_compile_definitions(CloudTune PRIVATE CLOUDTUNE_VERSION="${PROJECT_VERSION}")
target_include_directories(CloudTune PRIVATE vendor/qrcodegen)
target_link_libraries(CloudTune PRIVATE borealis netease_library ncm_runtime)
if(PLATFORM_SWITCH)
  target_sources(CloudTune PRIVATE ${BOREALIS_LIBRARY}/lib/platforms/switch/switch_wrapper.c)
  target_link_libraries(CloudTune PRIVATE ncm_ipc_client)
  add_custom_target(CloudTune.nro
    COMMAND ${NX_NACPTOOL_EXE} --create "CloudTune NX" "CloudTune NX contributors" "${PROJECT_VERSION}" CloudTune.nacp
    COMMAND ${NX_ELF2NRO_EXE} $<TARGET_FILE:CloudTune> CloudTune.nro
      --nacp=CloudTune.nacp --romfsdir=${CMAKE_CURRENT_SOURCE_DIR}/resources
    DEPENDS CloudTune
    WORKING_DIRECTORY ${CMAKE_BINARY_DIR}
    VERBATIM)
else()
  enable_testing()
  add_executable(ncm-library-tests tests/library_tests.cpp)
  target_link_libraries(ncm-library-tests PRIVATE netease_library)
  add_test(NAME library COMMAND ncm-library-tests)
  add_test(NAME queue-storage COMMAND ncm-queue-replace-test)
  add_executable(ncm-api-tests tests/api_tests.cpp)
  target_link_libraries(ncm-api-tests PRIVATE netease_api)
  add_test(NAME api COMMAND ncm-api-tests)
  add_executable(ncm-api-probe tools/api_probe.cpp)
  target_link_libraries(ncm-api-probe PRIVATE netease_api)
  add_executable(ncm-model-tests tests/model_tests.cpp src/model.cpp)
  target_link_libraries(ncm-model-tests PRIVATE netease_library)
  add_test(NAME model COMMAND ncm-model-tests)
  add_executable(ncm-storage-tests tests/storage_tests.cpp)
  target_link_libraries(ncm-storage-tests PRIVATE netease_api)
  add_test(NAME storage COMMAND ncm-storage-tests)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(ncm-airplay-audio-tests tests/airplay_audio_tests.cpp)
  target_link_libraries(ncm-airplay-audio-tests PRIVATE ncm_engine)
  add_test(NAME airplay-audio-handoff
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/airplay_audio_tests.py"
      $<TARGET_FILE:ncm-airplay-audio-tests>)
  set_tests_properties(airplay-audio-handoff PROPERTIES TIMEOUT 60)
  add_executable(ncm-airplay-codec-tests tests/airplay_codec_tests.cpp)
  target_link_libraries(ncm-airplay-codec-tests PRIVATE cloudtune_airplay)
  add_test(NAME airplay-codec COMMAND ncm-airplay-codec-tests)
  add_executable(ncm-airplay-mdns-tests tests/airplay_mdns_tests.cpp)
  target_link_libraries(ncm-airplay-mdns-tests PRIVATE cloudtune_airplay)
  add_test(NAME airplay-mdns COMMAND ncm-airplay-mdns-tests)
  set_tests_properties(airplay-mdns PROPERTIES TIMEOUT 15)
  add_executable(ncm-airplay-runtime-tests tests/airplay_runtime_tests.cpp)
  target_link_libraries(ncm-airplay-runtime-tests PRIVATE ncm_runtime)
  add_test(NAME airplay-runtime COMMAND ncm-airplay-runtime-tests)
  set_tests_properties(airplay-runtime PROPERTIES TIMEOUT 30)
  add_executable(ncm-airplay-receiver-probe tests/airplay_receiver_probe.cpp)
  target_link_libraries(ncm-airplay-receiver-probe PRIVATE cloudtune_airplay)
  add_test(NAME airplay-receiver
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/airplay_receiver_tests.py"
      $<TARGET_FILE:ncm-airplay-receiver-probe>)
  set_tests_properties(airplay-receiver PROPERTIES TIMEOUT 90)
  if(CLOUDTUNE_BUILD_OVERLAY_PREVIEW)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME overlay-preview-service
      COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/overlay_preview_tests.py"
        --service $<TARGET_FILE:cloudtune-preview-service>)
    set_tests_properties(overlay-preview-service PROPERTIES TIMEOUT 90)
  endif()
endif()
